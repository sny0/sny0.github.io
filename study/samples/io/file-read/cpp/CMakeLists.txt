cmake_minimum_required(VERSION 3.20)
project(file_read_observation LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This sample uses the Windows API.")
endif()

add_executable(file_read_observation main.cpp)
target_compile_features(file_read_observation PRIVATE cxx_std_20)
set_target_properties(file_read_observation PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(file_read_observation PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
else()
    target_compile_options(file_read_observation PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
