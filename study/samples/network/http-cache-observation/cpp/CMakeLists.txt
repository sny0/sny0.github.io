cmake_minimum_required(VERSION 3.20)
project(http_cache_observation LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This sample uses Windows WinHTTP.")
endif()
add_executable(http_cache_observation main.cpp)
target_compile_features(http_cache_observation PRIVATE cxx_std_20)
target_link_libraries(http_cache_observation PRIVATE winhttp)
set_target_properties(http_cache_observation PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(http_cache_observation PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
else()
    target_compile_options(http_cache_observation PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
