cmake_minimum_required(VERSION 3.20)
project(https_observation LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This sample uses Windows WinHTTP.")
endif()
add_executable(https_observation main.cpp)
target_compile_features(https_observation PRIVATE cxx_std_20)
target_link_libraries(https_observation PRIVATE winhttp crypt32)
set_target_properties(https_observation PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(https_observation PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
else()
    target_compile_options(https_observation PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
