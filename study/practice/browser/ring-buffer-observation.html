リングバッファの満杯・折り返し・読み出し順を観察する ​

整数を一つ入れるか、一つ取り出すたびに、4つの枠と二つの位置を見比べます。読み取り位置 R は次に取り出す枠、書き込み位置 W は次に書き込む候補です。空と満杯を二つの位置で区別するため、一度に保持できるのは3件です。リングバッファとSPSCキュー

リングバッファの実験を新しいタブで開く

満杯になった枠を空けて使い回す ​

公開ページの操作欄では、通常の ArrayBuffer を使ってキュー本体を実行します。「最初に戻す」で R = W = 0 に戻してから、「入れる」「取り出す」で操作します。入れる候補は、書き込みが成功したときだけ次の値に進みます。

操作操作後の R操作後の W次に取り出せる順
10、20、30を入れる0310、20、30
40を入れようとする0310、20、30
10を取り出す1320、30
40を入れ直す1020、30、40
20を取り出す2030、40
50を入れる2130、40、50
3回取り出す11空

満杯で40を入れようとすると、位置も中身もそのままです。10を取り出した後は枠3に40を書けます。書き込み位置が3から0へ戻り、20を取り出してから50を入れると枠0が再利用されます。値が置かれる物理的な順番と、取り出す順番を画面で比べます。

最後にもう一度取り出すと空と表示されます。枠を取り出したときに使える場所として扱うのは、読み取り位置が進むためです。コードではバイトを消去せず、画面ではキューに含まれる枠の値だけを表示します。

付属サーバーは文書とWorkerのスクリプトに分離用の応答ヘッダーを付けます。この配信環境で、必要なAPIが使えるときにWorkerの実験ボタンが有効になります。公開ページでも上の操作欄は使えます。クロスオリジン分離と共有メモリ

Workerの実験では、同じキュー本体を SharedArrayBuffer で使います。書き手が整数1から12までを順に入れ、読み手は取り出した値の順序を確認します。満杯や空で進めないときは、そのWorkerが間を置いて試し直します。画面は二つのWorkerが完了してから、成功した件数と順序の照合結果を表示します。

コード ​

4枠のキューと読み書き位置 ​

js
const ReadIndex = 0;
const WriteIndex = 1;
const PayloadStart = 2;
const SlotCount = 4;
export const QueueBytes = (PayloadStart + SlotCount) * Int32Array.BYTES_PER_ELEMENT;

function next(index) {
  return (index + 1) % SlotCount;
}

export function createQueueBuffer(shared = false) {
  return shared ? new SharedArrayBuffer(QueueBytes) : new ArrayBuffer(QueueBytes);
}

export function tryEnqueue(buffer, value) {
  if (!Number.isInteger(value) || value < -2147483648 || value > 2147483647) {
    throw new RangeError("符号付き32ビット整数を指定してください。");
  }
  const cells = new Int32Array(buffer);
  const write = Atomics.load(cells, WriteIndex);
  const nextWrite = next(write);
  if (nextWrite === Atomics.load(cells, ReadIndex)) return null;

  cells[PayloadStart + write] = value;
  // Publish the written slot by advancing the write index.
  Atomics.store(cells, WriteIndex, nextWrite);
  return write;
}

export function tryDequeue(buffer) {
  const cells = new Int32Array(buffer);
  const read = Atomics.load(cells, ReadIndex);
  if (read === Atomics.load(cells, WriteIndex)) return null;

  const value = cells[PayloadStart + read];
  // Release this slot for reuse after reading it.
  Atomics.store(cells, ReadIndex, next(read));
  return { slot: read, value };
}

export function snapshotQueue(buffer) {
  // The interactive view owns a normal buffer. The live Workers own the shared queue.
  if (!(buffer instanceof ArrayBuffer)) {
    throw new TypeError("表示用のスナップショットには通常のArrayBufferを指定してください。");
  }
  const cells = new Int32Array(buffer);
  const read = Atomics.load(cells, ReadIndex);
  const write = Atomics.load(cells, WriteIndex);
  const slots = Array(SlotCount).fill(null);
  for (let index = read; index !== write; index = next(index)) {
    slots[index] = cells[PayloadStart + index];
  }
  return { read, write, slots };
}

import { createQueueBuffer, snapshotQueue, tryDequeue, tryEnqueue } from "./ring-buffer.js";

const SampleValues = [10, 20, 30, 40, 50, 60, 70, 80, 90];
const TransferCount = 12;
const QueueStatus = document.getElementById("queue-status");
const QueueLog = document.getElementById("queue-log");
const Slots = [...document.querySelectorAll("[data-slot]")];
const RunWorkers = document.getElementById("run-workers");
const WorkerStatus = document.getElementById("worker-status");
const WorkerFields = ["produced", "consumed", "order", "last", "full", "empty"];
const CanRunWorkers = globalThis.crossOriginIsolated === true &&
  (location.protocol === "http:" || location.protocol === "https:") &&
  typeof Worker === "function" && typeof SharedArrayBuffer === "function";

let queue = createQueueBuffer();
let nextValueIndex = 0;
let lastDequeued = null;

function setText(id, value) {
  document.getElementById(id).textContent = value;
}

function renderQueue({ read, write, slots }) {
  const size = slots.filter(value => value !== null).length;
  const capacity = slots.length - 1;
  const fifo = Array.from({ length: size }, (_, offset) => slots[(read + offset) % slots.length]);
  setText("read-index", read);
  setText("write-index", write);
  setText("queue-size", size);
  setText("queue-state", size === 0 ? "空" : size === capacity ? "満杯" : "未読あり");
  setText("queue-order", fifo.length ? fifo.join(" → ") : "なし");
  setText("next-value", SampleValues[nextValueIndex]);
  setText("last-dequeued", lastDequeued ?? "—");

  for (const [index, slot] of Slots.entries()) {
    slot.dataset.occupied = String(slots[index] !== null);
    slot.querySelector(".slot-value").textContent = slots[index] ?? "空き";
    slot.querySelector(".badge.read").hidden = index !== read;
    slot.querySelector(".badge.write").hidden = index !== write;
  }
}

function recordOperation(state, message, snapshot) {
  renderQueue(snapshot);
  QueueStatus.dataset.state = state;
  QueueStatus.textContent = message;
  const entry = document.createElement("li");
  entry.textContent = message;
  QueueLog.prepend(entry);
  if (QueueLog.children.length > 10) QueueLog.lastElementChild.remove();
}

function movedCursor(label, before, after) {
  return `${label} は ${before} → ${after}${after === 0 ? "（先頭へ折り返し）" : ""}。`;
}

function enqueue() {
  const value = SampleValues[nextValueIndex];
  const slot = tryEnqueue(queue, value);
  const snapshot = snapshotQueue(queue);
  if (slot === null) {
    recordOperation("full", `満杯のため ${value} を保留しました。R と W は変わりません。`, snapshot);
    return;
  }

  nextValueIndex = (nextValueIndex + 1) % SampleValues.length;
  recordOperation("enqueued", `${value} を枠 ${slot} に入れました。${movedCursor("W", slot, snapshot.write)}`, snapshot);
}

function dequeue() {
  const entry = tryDequeue(queue);
  const snapshot = snapshotQueue(queue);
  if (entry === null) {
    recordOperation("empty", "キューは空です。R と W は変わりません。", snapshot);
    return;
  }

  lastDequeued = entry.value;
  recordOperation("dequeued", `枠 ${entry.slot} から ${entry.value} を取り出しました。${movedCursor("R", entry.slot, snapshot.read)}`, snapshot);
}

function reset() {
  queue = createQueueBuffer();
  nextValueIndex = 0;
  lastDequeued = null;
  QueueLog.replaceChildren();
  recordOperation("ready", "初期状態に戻しました。R と W はどちらも 0 です。", snapshotQueue(queue));
}

async function observeWorkers() {
  const buffer = createQueueBuffer(true);
  const workers = [];
  let timeout;
  try {
    const results = ["producer", "consumer"].map(async role => {
      const worker = new Worker(new URL("./ring-worker.js", import.meta.url), { type: "module" });
      workers.push(worker);
      return new Promise((resolve, reject) => {
        worker.addEventListener("message", ({ data }) => {
          if (data?.role === role && data.result) {
            resolve(data.result);
          } else {
            reject(new Error(data?.error || "Workerから想定外の結果が届きました。"));
          }
        }, { once: true });
        worker.addEventListener("error", () => reject(new Error("Workerを読み込むか実行する途中で失敗しました。")), { once: true });
        worker.addEventListener("messageerror", () => reject(new Error("Workerの結果を読み取れませんでした。")), { once: true });
        worker.postMessage({ buffer, role, count: TransferCount });
      });
    });
    const deadline = new Promise((_, reject) => {
      timeout = setTimeout(() => reject(new Error("Workerの応答が時間内に届きませんでした。")), 8000);
    });
    return await Promise.race([Promise.all(results), deadline]);
  } finally {
    clearTimeout(timeout);
    for (const worker of workers) worker.terminate();
  }
}

async function runWorkers() {
  RunWorkers.disabled = true;
  for (const field of WorkerFields) setText(`worker-${field}`, "—");
  WorkerStatus.dataset.state = "pending";
  WorkerStatus.textContent = "2つのWorkerの完了を待っています。";
  try {
    const [producer, consumer] = await observeWorkers();
    setText("worker-produced", producer.transferred);
    setText("worker-consumed", consumer.transferred);
    setText("worker-order", consumer.ordered ? "1 → 12 の順" : "順序が不一致");
    setText("worker-last", consumer.last);
    setText("worker-full", producer.full);
    setText("worker-empty", consumer.empty);
    const complete = producer.transferred === TransferCount && consumer.transferred === TransferCount &&
      consumer.ordered && consumer.last === TransferCount;
    WorkerStatus.dataset.state = complete ? "observed" : "error";
    WorkerStatus.textContent = complete
      ? "12件を受け渡し、入れた順に取り出せました。"
      : "実行結果の件数か順序が想定と異なります。";
  } catch (error) {
    WorkerStatus.dataset.state = "error";
    WorkerStatus.textContent = error.message;
  } finally {
    RunWorkers.disabled = !CanRunWorkers;
  }
}

for (const [id, handler] of [["enqueue", enqueue], ["dequeue", dequeue], ["reset", reset]]) {
  const button = document.getElementById(id);
  button.addEventListener("click", handler);
  button.disabled = false;
}
renderQueue(snapshotQueue(queue));

const availability = document.getElementById("worker-availability");
availability.dataset.state = CanRunWorkers ? "available" : "unavailable";
availability.textContent = CanRunWorkers
  ? "このページでは共有メモリを使うWorkerを実行できます。"
  : "この配信環境では一手ずつ動かせます。Workerで試すには付属サーバーで開きます。";
document.getElementById("worker-note").hidden = CanRunWorkers;
RunWorkers.disabled = !CanRunWorkers;
RunWorkers.addEventListener("click", runWorkers);

書き手と読み手のWorker ​

js
import { QueueBytes, tryDequeue, tryEnqueue } from "./ring-buffer.js";

const WaitLimitMs = 6000;
const MaxCount = 256;

function waitForRetry(deadline) {
  if (performance.now() >= deadline) {
    throw new Error("制限時間内にキューの操作を完了できませんでした。");
  }
  return new Promise(resolve => setTimeout(resolve, 0));
}

async function produce(buffer, count, deadline) {
  let full = 0;
  for (let value = 1; value <= count;) {
    if (tryEnqueue(buffer, value) === null) {
      full += 1;
      await waitForRetry(deadline);
    } else {
      value += 1;
    }
  }
  return { transferred: count, full };
}

async function consume(buffer, count, deadline) {
  let empty = 0;
  let ordered = true;
  let last = null;
  for (let expected = 1; expected <= count;) {
    const entry = tryDequeue(buffer);
    if (entry === null) {
      empty += 1;
      await waitForRetry(deadline);
    } else {
      last = entry.value;
      ordered = ordered && last === expected;
      expected += 1;
    }
  }
  return { transferred: count, empty, ordered, last };
}

self.addEventListener("message", async ({ data }) => {
  const { role, buffer, count } = data ?? {};
  try {
    if (typeof SharedArrayBuffer !== "function" ||
        !(buffer instanceof SharedArrayBuffer) || buffer.byteLength !== QueueBytes) {
      throw new Error("実測用の共有キューバッファが必要です。");
    }
    if (!Number.isInteger(count) || count < 1 || count > MaxCount) {
      throw new Error(`試す件数は1〜${MaxCount}の整数で指定してください。`);
    }
    if (role !== "producer" && role !== "consumer") {
      throw new Error("生産側または消費側の役割を指定してください。");
    }
    const run = role === "producer" ? produce : consume;
    const result = await run(buffer, count, performance.now() + WaitLimitMs);
    self.postMessage({ role, result });
  } catch (error) {
    self.postMessage({ role, error: error.message });
  }
}, { once: true });

ローカル配信の分離用ヘッダー ​

python
"""Serve the ring-buffer lab with isolation headers on IPv4 loopback."""

from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path


HOST = "127.0.0.1"
PORT = 4188
TYPES = {
    "index.html": "text/html; charset=utf-8",
    "styles.css": "text/css; charset=utf-8",
    "main.js": "text/javascript; charset=utf-8",
    "ring-buffer.js": "text/javascript; charset=utf-8",
    "ring-worker.js": "text/javascript; charset=utf-8",
}


def page_routes(directory):
    routes = {
        f"/{name}": (content_type, (directory / name).read_bytes())
        for name, content_type in TYPES.items()
    }
    routes["/"] = routes["/index.html"]
    return routes


class Handler(BaseHTTPRequestHandler):
    def setup(self):
        self.request.settimeout(5)
        super().setup()

    def log_message(self, format, *args):
        # The lab does not log arbitrary request targets.
        pass

    def do_GET(self):
        self.respond(include_body=True)

    def do_HEAD(self):
        self.respond(include_body=False)

    def respond(self, include_body):
        # Use the raw request target so a normalized path is not an alias.
        target = self.requestline.split()[1]
        expected_host = f"{HOST}:{self.server.server_port}"
        if self.headers.get_all("Host") != [expected_host]:
            status, content_type, body = 400, "text/plain; charset=utf-8", b"Bad Host\n"
        elif target not in self.server.routes:
            status, content_type, body = 404, "text/plain; charset=utf-8", b"Not Found\n"
        else:
            status = 200
            content_type, body = self.server.routes[target]

        self.send_response(status)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(len(body)))
        self.send_header("Cache-Control", "no-store")
        self.send_header("X-Content-Type-Options", "nosniff")
        if status == 200:
            self.send_header("Cross-Origin-Embedder-Policy", "require-corp")
            if target in {"/", "/index.html"}:
                self.send_header("Cross-Origin-Opener-Policy", "same-origin")
        self.end_headers()
        if include_body:
            self.wfile.write(body)


class LabServer(ThreadingHTTPServer):
    allow_reuse_address = False

    def __init__(self, port, routes):
        self.routes = routes
        super().__init__((HOST, port), Handler)


def main():
    try:
        routes = page_routes(Path(__file__).resolve().parent)
        with LabServer(PORT, routes) as server:
            print(f"http://{HOST}:{PORT}/", flush=True)
            try:
                server.serve_forever()
            except KeyboardInterrupt:
                pass
    except OSError as error:
        raise SystemExit(f"Could not start the lab: {error}") from error


if __name__ == "__main__":
    main()
