Workerの競合とAtomicsの待機・通知を観察する ​

同じカウンターを二つのWorkerから1ずつ増やします。読み取りと書き戻しを分けたときと、Atomics.add() でまとめたときに、何回分の更新が残るかを比べます。そのあと、フラグと通知を使ってWorker間で値を渡します。共有メモリと各操作の仕組みは辞典記事で確認できます。Web Workerと共有メモリの同期

WorkerとAtomicsの実験を新しいタブで開く

公開ページで操作の順序をたどる ​

公開ページのステップ実験は、ボタンを押すたびに、あらかじめ選んだ順序でモデルの状態を一つ進めます。各実験はリセットして繰り返せます。

実験操作の順序最後に確認する値
読み取りと書き戻しを分けるAが0を読む → Bが0を読む → Aが1を書く → Bが1を書くカウンターは1
不可分に加算するAが加算して旧値0を得る → Bが加算して旧値1を得るカウンターは2
フラグでデータの準備を伝える受信側が待つ → 送信側が42を書く → フラグを1にする → 通知する → 受信側がフラグと値を読む受信側の値は42

最初の実験では、Atomics.load() と Atomics.store() がそれぞれアトミックでも、二つの呼び出しの間に別の処理が入ります。二番目では Atomics.add() が読み取り・加算・書き戻しを一つの操作として行います。ECMAScriptのAtomics

付属サーバーは文書とWorkerのスクリプトに、クロスオリジン分離に必要なヘッダーを付けます。公開ページでは現在の実行環境を表示し、分離と必要なAPIがそろうときに実測を有効にします。クロスオリジン分離と共有メモリ

実測では、同じ SharedArrayBuffer から Int32Array のビューを作ります。添字0がカウンター、1が準備フラグ、2がデータです。実験ごとに新しいバッファを使います。

  1. 読み取りと書き戻しを分ける実験では、ページ側が両方の読み取り結果を受け取ってから、書き戻しを指示します。両方が0を読んだ状態を作るため、最後に残る値は1です。
  2. 不可分な加算では、どちらのWorkerが先に実行されても、二つの戻り値は0と1、最後に残る値は2になります。Worker名と戻り値の対応は実行順で変わります。
  3. データの受け渡しでは、送信側が値42を書き、フラグを1にしてから通知します。受信側はフラグを確認してから値を読みます。待機が始まる前に通知されることもあるため、通知された件数と受信側の値を分けて確認します。

Atomics.notify() の戻り値は、その呼び出しで通知された待機の数です。待機に入る直前にフラグが変わると Atomics.wait() は not-equal を返します。受信側が最初の確認でフラグ1を読んだ場合は、待機する必要がありません。通知で ok が返った場合も、受信側はフラグを読み直してからデータを使います。ECMAScriptのAtomics.wait、Atomics.notify

const ModelSteps = {
  split: [
    ["A が load で 0 を読み取る。", {a: 0}],
    ["B も load で 0 を読み取る。", {b: 0}],

function setupModel(panel) {
  const steps = ModelSteps[panel.dataset.model];
  const fields = Array.from(panel.querySelectorAll("[data-field]"));
  const initialValues = fields.map(field => field.textContent);
  const next = panel.querySelector("[data-step]");
  const reset = panel.querySelector("[data-reset]");
  const status = panel.querySelector("[data-status]");
  const log = panel.querySelector("[data-log]");
  let position = 0;

  function updateControls(message = "") {
    next.disabled = position === steps.length;
    reset.disabled = position === 0;
    status.dataset.state = position === steps.length ? "complete" : "ready";
    status.textContent = `${position} / ${steps.length} 手順${message ? `。 ${message}` : ""}`;
  }

  next.addEventListener("click", () => {
    const [message, changes] = steps[position];
    for (const field of fields) {
      const value = changes[field.dataset.field];
      if (value !== undefined) field.textContent = value;
    }
    const item = document.createElement("li");
    item.textContent = message;
    log.append(item);
    position += 1;
    updateControls(message);
  });
  reset.addEventListener("click", () => {
    fields.forEach((field, index) => { field.textContent = initialValues[index]; });
    log.replaceChildren();
    position = 0;
    updateControls();
  });
  updateControls();
}

const Counter = 0;
const Isolated = window.crossOriginIsolated === true;
const CanRunWorkers = Isolated &&
  ["http:", "https:"].includes(location.protocol) &&
  typeof Worker === "function" &&
  typeof SharedArrayBuffer === "function" &&
  typeof window.Atomics?.wait === "function";

function observeWorkers(mode) {
  return new Promise((resolve, reject) => {
    const buffer = new SharedArrayBuffer(3 * Int32Array.BYTES_PER_ELEMENT);
    const values = new Int32Array(buffer);
    const workers = [];
    const requests = new Map();
    let nextId = 0;
    let finished = false;
    const timer = setTimeout(() => finish(new Error("Workerの応答が時間内に届きませんでした。")), 8000);

    function finish(error, result) {
      if (finished) return;
      finished = true;
      clearTimeout(timer);
      for (const worker of workers) worker.terminate();
      if (error) {
        for (const request of requests.values()) request.reject(error);
        requests.clear();
        reject(error);
      } else {
        resolve(result);
      }
    }

    function openWorker() {
      const worker = new Worker("./worker.js");
      workers.push(worker);
      worker.addEventListener("error", () => finish(new Error("Workerを読み込むか実行する途中で失敗しました。")));
      worker.addEventListener("messageerror", () => finish(new Error("Workerのメッセージを読み取れませんでした。")));
      worker.addEventListener("message", ({data}) => {
        const request = requests.get(data?.id);
        if (!request || request.worker !== worker || request.action !== data.action) {
          finish(new Error("Workerから想定外のメッセージが届きました。"));
          return;
        }
        if (data.type === "waiting" && request.onWaiting) {
          const publish = request.onWaiting;
          request.onWaiting = undefined;
          publish();
        } else if (data.type === "result") {
          requests.delete(data.id);
          request.resolve(data);
        } else {
          finish(new Error(data.message || "Workerの操作に失敗しました。"));
        }
      });
      return worker;
    }

    function command(worker, action, onWaiting) {
      return new Promise((resolve, reject) => {
        const id = ++nextId;
        requests.set(id, {worker, action, onWaiting, resolve, reject});
        try {
          worker.postMessage({id, action, buffer});
        } catch (error) {
          requests.delete(id);
          reject(error);
        }
      });
    }

    async function execute() {
      const a = openWorker();
      const b = openWorker();
      if (mode === "split") {
        const [aRead, bRead] = await Promise.all([command(a, "read"), command(b, "read")]);
        // Both reads have completed before either Worker can store.
        await Promise.all([command(a, "write"), command(b, "write")]);
        return {a: aRead.value, b: bRead.value, counter: Atomics.load(values, Counter)};
      }
      if (mode === "add") {
        const [aAdd, bAdd] = await Promise.all([command(a, "add"), command(b, "add")]);
        return {a: aAdd.before, b: bAdd.before, counter: Atomics.load(values, Counter)};
      }

      let publication;
      const received = await command(a, "wait", () => {
        // The message confirms the reader saw 0. It may still be about to call wait.
        publication = command(b, "publish");
        publication.catch(finish);
      });
      if (!publication) throw new Error("フラグ0を確認した読み手のメッセージが届きませんでした。");
      const sent = await publication;
      return {waitResult: received.status, woken: sent.woken, flag: received.flag, read: received.payload};
    }

    execute().then(result => finish(null, result), finish);
  });
}

async function runWorkerCase(panel) {
  const mode = panel.dataset.worker;
  for (const button of WorkerButtons) button.disabled = true;
  showWorkerResult(panel, "pending", "Workerの応答を待っています。");
  try {
    const values = await observeWorkers(mode);
    if (mode !== "wait") {
      showWorkerResult(panel, "observed", `共有カウンターは ${values.counter} です。`, values);
    } else if (values.flag === 1) {
      showWorkerResult(panel, "observed", `読み手はフラグ1を確認し、${values.read}を取得しました。`, values);
    } else {
      showWorkerResult(panel, "observed", "読み手の待機時間内にフラグが1にならず、データの読み取りは行いませんでした。", values);
    }
  } catch (error) {
    showWorkerResult(panel, "error", error.message);
  } finally {
    for (const button of WorkerButtons) button.disabled = !CanRunWorkers;
  }
}

  button.addEventListener("click", () => void runWorkerCase(panel));
}

Workerで共有メモリを読み書きする ​

js
"use strict";

const Counter = 0;
const Ready = 1;
const Payload = 2;
const SlotBytes = 3 * Int32Array.BYTES_PER_ELEMENT;
const WaitLimitMs = 2000;

let snapshot = null;

function waitForPayload(slots, id) {
  let flag = Atomics.load(slots, Ready);
  let status = "already-ready";
  const deadline = performance.now() + WaitLimitMs;
  if (flag === 0) {
    self.postMessage({ id, type: "waiting", action: "wait", value: flag });
  }

  while (flag === 0) {
    const remaining = deadline - performance.now();
    if (remaining <= 0) {
      status = "timed-out";
      break;
    }
    status = Atomics.wait(slots, Ready, 0, remaining);
    flag = Atomics.load(slots, Ready);
  }

  // Read the payload only after observing the writer's ready flag.
  return { flag, payload: flag === 1 ? slots[Payload] : null, status };
}

function execute(action, slots, id) {
  switch (action) {
    case "read":
      snapshot = Atomics.load(slots, Counter);
      return { value: snapshot };
    case "write": {
      if (snapshot === null) throw new Error("先にカウンターを読み取ってください。");
      const value = Atomics.store(slots, Counter, snapshot + 1);
      snapshot = null;
      return { value };
    }
    case "add": {
      const before = Atomics.add(slots, Counter, 1);
      return { before, value: before + 1 };
    }
    case "wait":
      return waitForPayload(slots, id);
    case "publish": {
      const payload = 42;
      slots[Payload] = payload;
      const flag = Atomics.store(slots, Ready, 1);
      const woken = Atomics.notify(slots, Ready, 1);
      return { flag, payload, woken };
    }
    default:
      throw new Error("実験で使う操作を指定してください。");
  }
}

self.addEventListener("message", ({ data }) => {
  const { id, action, buffer } = data ?? {};
  try {
    if (!(buffer instanceof SharedArrayBuffer) || buffer.byteLength !== SlotBytes) {
      throw new Error("12バイトの共有領域が必要です。");
    }
    const slots = new Int32Array(buffer);
    self.postMessage({ id, type: "result", action, ...execute(action, slots, id) });
  } catch (error) {
    self.postMessage({ id, type: "error", action, message: error.message });
  }
});

ローカル配信で分離用のヘッダーを付ける ​

python
"""Serve the Atomics lab with isolation headers on IPv4 loopback."""

from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path


HOST = "127.0.0.1"
PORT = 4187
TYPES = {
    "index.html": "text/html; charset=utf-8",
    "styles.css": "text/css; charset=utf-8",
    "main.js": "text/javascript; charset=utf-8",
    "worker.js": "text/javascript; charset=utf-8",
}


def page_routes(directory):
    routes = {
        f"/{name}": (content_type, (directory / name).read_bytes())
        for name, content_type in TYPES.items()
    }
    routes["/"] = routes["/index.html"]
    return routes


class Handler(BaseHTTPRequestHandler):
    def setup(self):
        self.request.settimeout(5)
        super().setup()

    def log_message(self, format, *args):
        # The lab does not log arbitrary request targets.
        pass

    def do_GET(self):
        self.respond(include_body=True)

    def do_HEAD(self):
        self.respond(include_body=False)

    def respond(self, include_body):
        # Use the raw request target so a normalized path is not an alias.
        target = self.requestline.split()[1]
        expected_host = f"{HOST}:{self.server.server_port}"
        if self.headers.get_all("Host") != [expected_host]:
            status, content_type, body = 400, "text/plain; charset=utf-8", b"Bad Host\n"
        elif target not in self.server.routes:
            status, content_type, body = 404, "text/plain; charset=utf-8", b"Not Found\n"
        else:
            status = 200
            content_type, body = self.server.routes[target]

        self.send_response(status)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(len(body)))
        self.send_header("Cache-Control", "no-store")
        self.send_header("X-Content-Type-Options", "nosniff")
        if status == 200:
            if target in {"/", "/index.html"}:
                self.send_header("Cross-Origin-Opener-Policy", "same-origin")
            if target in {"/", "/index.html", "/worker.js"}:
                self.send_header("Cross-Origin-Embedder-Policy", "require-corp")
        self.end_headers()
        if include_body:
            self.wfile.write(body)


class LabServer(ThreadingHTTPServer):
    allow_reuse_address = False

    def __init__(self, port, routes):
        self.routes = routes
        super().__init__((HOST, port), Handler)


def main():
    try:
        routes = page_routes(Path(__file__).resolve().parent)
        with LabServer(PORT, routes) as server:
            print(f"http://{HOST}:{PORT}/", flush=True)
            try:
                server.serve_forever()
            except KeyboardInterrupt:
                pass
    except OSError as error:
        raise SystemExit(f"Could not start the lab: {error}") from error


if __name__ == "__main__":
    main()
