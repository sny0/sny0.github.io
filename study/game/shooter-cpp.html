シューティング実装例 — C++17とraylib ​

設計の考え方と共通仕様をC++へ落とした例です。C++17を使い、キーボードと円の描画だけをraylib 5.5に任せます。ゲームの状態とルールはraylibに依存しません。Unity版と責務を比べながら読めます。

ファイルの役割 ​

ファイル役割
shooter_model.hpp自機・弾・敵・HP・スコア、1tickのルール
main.cpp入力、実時間から固定tickへの変換、ウィンドウ、描画
CMakeLists.txtModelの検査とraylib版のビルド設定
model_checks.cpp画面を開かずにゲームの不変条件を確認

Modelを画面なしで実行できるため、発射間隔・衝突・再開を調べるたびに、ウィンドウを操作して敵を狙う必要がありません。ただし、Modelの検査だけで描画・キー入力・OS上の動作まで検証できるわけではありません。

起動する ​

ファイルを保存したディレクトリで実行します。CMake 3.20以上、C/C++コンパイラ、対応するビルドツール、依存取得用のGitが必要です。raylib 5.5を既にCMakeから見つけられる環境ではそのライブラリを使い、ない場合はCMake設定時に5.5のソースを取得してビルドします。初回の依存取得にはネットワーク接続が必要です。Linux等では、そのOS向けのウィンドウ・OpenGL関連の開発用依存も必要になります。

WindowsのMinGW Makefilesで確認する場合は、英数字だけの作業フォルダを使ってください。今回のMinGW環境では、日本語を含む絶対パスで入力ファイルを見つけられずにビルドが止まる問題がありました。他のコンパイラやGeneratorも一律に同じ制約という意味ではありません。

sh
cmake -S . -B build
cmake --build build --config Release
ctest --test-dir build -C Release --output-on-failure

実行ファイル名や、画面なしで検査する設定は下のCMakeとREADMEで確認できます。Visual Studioのような複数構成のGeneratorでは、実行ファイルがbuild/Release/等に置かれます。単一構成のGeneratorでは通常build/です。

WindowsでVisual Studio 2022を使う場合は、最初の設定を次のように指定できます。今回の描画ライブラリ込みの確認も、この構成を使いました。

sh
cmake -S . -B build-vs -G "Visual Studio 17 2022" -A x64
cmake --build build-vs --config Release
ctest --test-dir build-vs -C Release --output-on-failure
./build-vs/Release/shooter.exe
sh
# 単一構成の例。Windowsなら build/shooter.exe
./build/shooter

画面を開かずModelだけを検査する場合は、raylibの取得を省略できます。

sh
cmake -S . -B build-model -DSHOOTER_BUILD_GAME=OFF
cmake --build build-model --config Release
ctest --test-dir build-model -C Release --output-on-failure

操作は矢印/WASD、Space、Rです。敵の出現位置は循環するので、自機を動かして敵の下へ入り、弾が当たる位置を探します。

Modelの全コード ​

Worldが状態を持ち、Tickで1/60秒分だけ進めます。移動を正規化する理由、衝突の順番、消去を最後にまとめる理由は設計ページで説明しています。

cpp
#ifndef SNY0_SHOOTER_MODEL_HPP
#define SNY0_SHOOTER_MODEL_HPP

#include <algorithm>
#include <array>
#include <cmath>
#include <cstddef>
#include <vector>

namespace shooter {

inline constexpr int kWidth = 640;
inline constexpr int kHeight = 720;
inline constexpr float kTickSeconds = 1.0f / 60.0f;
inline constexpr float kPlayerRadius = 18.0f;
inline constexpr float kPlayerSpeed = 280.0f;
inline constexpr float kBulletRadius = 5.0f;
inline constexpr float kBulletSpeed = 520.0f;
inline constexpr float kEnemyRadius = 16.0f;
inline constexpr float kEnemySpeed = 110.0f;
inline constexpr int kShotCooldownTicks = 9;     // 0.15 seconds at 60 Hz
inline constexpr int kEnemySpawnTicks = 39;      // 0.65 seconds at 60 Hz
inline constexpr int kInvulnerabilityTicks = 48; // 0.8 seconds at 60 Hz
inline constexpr std::size_t kMaxBullets = 64;
inline constexpr std::size_t kMaxEnemies = 32;
inline constexpr std::array<float, 6> kEnemySpawnX{64, 160, 256, 352, 448, 544};

struct Vec2 {
    float x = 0.0f;
    float y = 0.0f;
};

struct FrameInput {
    float moveX = 0.0f;
    float moveY = 0.0f;
    bool shoot = false;
    bool restart = false;
};

struct Player {
    Vec2 position{320.0f, 650.0f};
    int hp = 3;
    int invulnerabilityTicks = 0;
};

struct Bullet {
    Vec2 position;
    bool alive = true;
};

struct Enemy {
    Vec2 position;
    bool alive = true;
};

inline bool CirclesOverlap(Vec2 a, float radiusA, Vec2 b, float radiusB) {
    const float dx = a.x - b.x;
    const float dy = a.y - b.y;
    const float radii = radiusA + radiusB;
    return dx * dx + dy * dy <= radii * radii;
}

// Value state for this small teaching example. The adapter only mutates it via
// Tick/Reset, and its renderer accepts const World&. Checks arrange valid state
// directly to exercise collisions without waiting for a particular game scene.
struct World {
    Player player;
    std::vector<Bullet> bullets;
    std::vector<Enemy> enemies;
    int score = 0;
    bool gameOver = false;
    int shotCooldownTicks = 0;
    int spawnCountdownTicks = kEnemySpawnTicks;
    std::size_t nextSpawnIndex = 0;

    World() { Reset(); }

    void Reset() {
        player = Player{};
        bullets.clear();
        enemies.clear();
        bullets.reserve(kMaxBullets);
        enemies.reserve(kMaxEnemies);
        score = 0;
        gameOver = false;
        shotCooldownTicks = 0;
        spawnCountdownTicks = kEnemySpawnTicks;
        nextSpawnIndex = 0;
    }

    // No clocks, input devices, graphics, random generator or variable dt here.
    // Every call advances exactly one simulation tick, unless frozen/reset.
    void Tick(FrameInput input) {
        if (input.restart) {
            Reset();
            return;
        }
        if (gameOver) return;

        if (shotCooldownTicks > 0) --shotCooldownTicks;
        if (player.invulnerabilityTicks > 0) --player.invulnerabilityTicks;

        const float lengthSquared = input.moveX * input.moveX + input.moveY * input.moveY;
        if (lengthSquared > 0.0f) {
            const float scale = kPlayerSpeed * kTickSeconds / std::sqrt(lengthSquared);
            player.position.x += input.moveX * scale;
            player.position.y += input.moveY * scale;
        }
        player.position.x = std::clamp(player.position.x, kPlayerRadius, kWidth - kPlayerRadius);
        player.position.y = std::clamp(player.position.y, kPlayerRadius, kHeight - kPlayerRadius);

        if (input.shoot && shotCooldownTicks == 0) {
            if (bullets.size() < kMaxBullets)
                bullets.push_back({{player.position.x, player.position.y - 24.0f}, true});
            // A skipped shot still consumes its scheduled cooldown.
            shotCooldownTicks = kShotCooldownTicks;
        }

        --spawnCountdownTicks;
        if (spawnCountdownTicks == 0) {
            const float x = kEnemySpawnX[nextSpawnIndex];
            nextSpawnIndex = (nextSpawnIndex + 1) % kEnemySpawnX.size();
            if (enemies.size() < kMaxEnemies)
                enemies.push_back({{x, -kEnemyRadius}, true});
            // A skipped spawn still consumes its interval and x-cycle entry.
            spawnCountdownTicks = kEnemySpawnTicks;
        }

        // New objects also move in the tick in which they are created.
        for (auto& bullet : bullets)
            if (bullet.alive) bullet.position.y -= kBulletSpeed * kTickSeconds;
        for (auto& enemy : enemies)
            if (enemy.alive) enemy.position.y += kEnemySpeed * kTickSeconds;

        // Bullet/enemy collisions precede enemy/player contacts.
        for (auto& bullet : bullets) {
            if (!bullet.alive) continue;
            for (auto& enemy : enemies) {
                if (!enemy.alive) continue;
                if (CirclesOverlap(bullet.position, kBulletRadius, enemy.position, kEnemyRadius)) {
                    bullet.alive = false;
                    enemy.alive = false;
                    score += 100;
                    break; // One bullet cannot score against several enemies.
                }
            }
        }
        for (auto& enemy : enemies) {
            if (!enemy.alive) continue;
            if (CirclesOverlap(enemy.position, kEnemyRadius, player.position, kPlayerRadius)) {
                enemy.alive = false; // Contact removes the enemy even while invulnerable.
                if (player.invulnerabilityTicks == 0 && player.hp > 0) {
                    --player.hp;
                    player.invulnerabilityTicks = kInvulnerabilityTicks;
                    if (player.hp == 0) gameOver = true;
                }
            }
        }

        for (auto& bullet : bullets)
            if (bullet.position.y + kBulletRadius < 0.0f) bullet.alive = false;
        for (auto& enemy : enemies)
            if (enemy.position.y - kEnemyRadius > kHeight) enemy.alive = false;

        // Mark first, erase once. References used by the loops above are gone.
        bullets.erase(std::remove_if(bullets.begin(), bullets.end(),
                      [](const Bullet& b) { return !b.alive; }), bullets.end());
        enemies.erase(std::remove_if(enemies.begin(), enemies.end(),
                      [](const Enemy& e) { return !e.alive; }), enemies.end());
    }
};

} // namespace shooter

#endif

入力・ゲームループ・描画の全コード ​

キーを読む場所でFrameInputを作り、状態を進める場所へ渡します。描画は状態を読み取ります。Game Over中もこの外側のループを回すため、再開の入力を受け付けられます。

cpp
#include "shooter_model.hpp"
#include "raylib.h"

#include <algorithm>
#include <cmath>
#include <cstdio>
#include <cstring>

namespace {

shooter::FrameInput ReadInput() {
    shooter::FrameInput input;
    input.moveX = static_cast<float>((IsKeyDown(KEY_RIGHT) || IsKeyDown(KEY_D)) -
                                    (IsKeyDown(KEY_LEFT) || IsKeyDown(KEY_A)));
    input.moveY = static_cast<float>((IsKeyDown(KEY_DOWN) || IsKeyDown(KEY_S)) -
                                    (IsKeyDown(KEY_UP) || IsKeyDown(KEY_W)));
    input.shoot = IsKeyDown(KEY_SPACE);
    return input;
}

void DrawWorld(const shooter::World& world) {
    ClearBackground(Color{12, 19, 35, 255});
    for (int i = 0; i < 28; ++i)
        DrawPixel((i * 97 + 31) % shooter::kWidth, (i * 137 + 17) % shooter::kHeight,
                  Color{70, 87, 116, 255});

    for (const auto& bullet : world.bullets)
        DrawCircleV({bullet.position.x, bullet.position.y}, shooter::kBulletRadius, YELLOW);
    for (const auto& enemy : world.enemies) {
        DrawCircleV({enemy.position.x, enemy.position.y}, shooter::kEnemyRadius, RED);
        DrawCircleLines(static_cast<int>(enemy.position.x), static_cast<int>(enemy.position.y),
                        shooter::kEnemyRadius, MAROON);
    }
    const auto& p = world.player;
    if (p.invulnerabilityTicks == 0 || (p.invulnerabilityTicks / 4) % 2 == 0) {
        DrawCircleV({p.position.x, p.position.y}, shooter::kPlayerRadius, SKYBLUE);
        DrawTriangle({p.position.x, p.position.y - 20.0f},
                     {p.position.x - 12.0f, p.position.y + 10.0f},
                     {p.position.x + 12.0f, p.position.y + 10.0f}, BLUE);
    }

    DrawText(TextFormat("HP %d", p.hp), 16, 16, 24, GREEN);
    DrawText(TextFormat("SCORE %d", world.score), 420, 16, 24, YELLOW);
    DrawText("ARROWS / WASD: MOVE    SPACE: SHOT    R: RESTART", 16, 692, 16, LIGHTGRAY);
    if (world.gameOver) {
        DrawRectangle(90, 285, 460, 135, Fade(BLACK, 0.85f));
        DrawText("GAME OVER", 190, 305, 36, RAYWHITE);
        DrawText("Press R to restart", 200, 365, 24, LIGHTGRAY);
    }
}

} // namespace

int main(int argc, char** argv) {
    const bool smoke = argc > 1 && std::strcmp(argv[1], "--smoke") == 0;
    if (smoke) SetConfigFlags(FLAG_WINDOW_HIDDEN);
    InitWindow(shooter::kWidth, shooter::kHeight, "Sny0 - Vertical Shooter (C++17 / raylib 5.5)");
    if (!IsWindowReady()) return 1;
    SetTargetFPS(120); // Rendering rate; the Model below still advances at 60 Hz.

    shooter::World world;
    constexpr double tick = 1.0 / 60.0;
    constexpr int maxTicksPerFrame = 5;
    double accumulator = 0.0;
    bool restartPending = false;
    int renderedFrames = 0;

    while (!WindowShouldClose() && (!smoke || renderedFrames < 5)) {
        auto input = ReadInput();
        restartPending = restartPending || IsKeyPressed(KEY_R);
        accumulator += std::min(static_cast<double>(GetFrameTime()), 0.1);

        int steps = 0;
        while (accumulator >= tick && steps < maxTicksPerFrame) {
            input.restart = restartPending;
            restartPending = false; // Consume the edge once, not once per catch-up tick.
            world.Tick(input);
            accumulator -= tick;
            ++steps;
        }
        if (accumulator >= tick) {
            // Explicit overload policy: discard whole unprocessed ticks, keep
            // only the fractional remainder. Simulation falls behind wall time.
            accumulator = std::fmod(accumulator, tick);
        }

        BeginDrawing();
        DrawWorld(world); // Latest state; no interpolation or extrapolation here.
        EndDrawing();
        ++renderedFrames;
    }
    CloseWindow();
    if (smoke) {
        if (renderedFrames != 5) return 1;
        std::puts("PASS: hidden raylib 5.5 window rendered 5 frames");
    }
    return 0;
}

Modelの検査コード ​

画面なしで、仕様上起きてほしくない変化を確認します。例えば「弾1個に敵が重なっていても加点は1回」「Game Over後は状態が進まない」「Rで初期状態へ戻る」を検査します。

cpp
#include "shooter_model.hpp"

#include <cmath>
#include <cstdio>
#include <cstdlib>

namespace {

int checks = 0;

void Check(bool condition, const char* message) {
    ++checks;
    if (!condition) {
        std::fprintf(stderr, "FAILED: %s\n", message);
        std::exit(EXIT_FAILURE);
    }
}

bool Near(float a, float b) { return std::fabs(a - b) < 0.001f; }

void Advance(shooter::World& w, int count, shooter::FrameInput input = {}) {
    for (int i = 0; i < count; ++i) w.Tick(input);
}

// Arrange an overlap after this tick's movement phase.
shooter::Enemy ContactEnemy(const shooter::World& w) {
    return {{w.player.position.x,
             w.player.position.y - shooter::kEnemySpeed * shooter::kTickSeconds}, true};
}

shooter::Bullet BulletAtAfterMove(float x, float y) {
    return {{x, y + shooter::kBulletSpeed * shooter::kTickSeconds}, true};
}

void InitialMovementAndBounds() {
    shooter::World w;
    Check(w.player.hp == 3 && w.score == 0 && !w.gameOver, "initial HP/score/state");
    Check(Near(w.player.position.x, 320) && Near(w.player.position.y, 650), "initial player center");
    Check(w.bullets.empty() && w.enemies.empty(), "initial object lists");
    w.Tick({1, 1, false, false});
    const float dx = w.player.position.x - 320;
    const float dy = w.player.position.y - 650;
    Check(Near(std::sqrt(dx * dx + dy * dy), shooter::kPlayerSpeed * shooter::kTickSeconds),
          "diagonal movement is normalized");
    Advance(w, 300, {1, 1, false, false});
    Check(Near(w.player.position.x, 622) && Near(w.player.position.y, 702), "right/bottom radius bounds");
    w.Reset();
    Advance(w, 300, {-1, -1, false, false});
    Check(Near(w.player.position.x, 18) && Near(w.player.position.y, 18), "left/top radius bounds");
    Check(shooter::CirclesOverlap({0, 0}, 5, {21, 0}, 16), "touching circles count as contact");
    Check(!shooter::CirclesOverlap({0, 0}, 5, {21.1f, 0}, 16), "separated circles do not contact");
}

void ShotAndSpawnTiming() {
    shooter::World w;
    const shooter::FrameInput heldShot{0, 0, true, false};
    w.Tick(heldShot);
    Check(w.bullets.size() == 1, "first held shot is immediate");
    Check(Near(w.bullets[0].position.y, 650 - 24 - 520 * shooter::kTickSeconds),
          "shot starts at playerY-24 and moves in its creation tick");
    Advance(w, 8, heldShot);
    Check(w.bullets.size() == 1, "no second shot before nine elapsed ticks");
    w.Tick(heldShot);
    Check(w.bullets.size() == 2, "second shot at tick 10");
    Advance(w, 9, heldShot);
    Check(w.bullets.size() == 3, "third shot at tick 19");

    w.Reset();
    Advance(w, 38);
    Check(w.enemies.empty(), "no spawn before 0.65 seconds");
    w.Tick({});
    Check(w.enemies.size() == 1 && Near(w.enemies[0].position.x, 64), "first spawn at tick 39");
    Check(Near(w.enemies[0].position.y, -16 + 110 * shooter::kTickSeconds), "spawn moves in creation tick");
    Advance(w, 39 * 5);
    Check(w.enemies.size() == 6, "six scheduled spawns");
    for (std::size_t i = 0; i < shooter::kEnemySpawnX.size(); ++i)
        Check(Near(w.enemies[i].position.x, shooter::kEnemySpawnX[i]), "spawn x sequence");
    Advance(w, 39);
    Check(w.enemies.size() == 7 && Near(w.enemies.back().position.x, 64), "spawn x sequence wraps");
}

void CollisionOrderAndSingleHit() {
    shooter::World w;
    w.bullets.push_back(BulletAtAfterMove(200, 200));
    w.enemies.push_back({{200, 200 - shooter::kEnemySpeed * shooter::kTickSeconds}, true});
    w.enemies.push_back(w.enemies.front());
    w.Tick({});
    Check(w.score == 100 && w.bullets.empty() && w.enemies.size() == 1,
          "one bullet cannot hit two overlapping enemies");

    w.Reset();
    w.bullets.push_back(BulletAtAfterMove(200, 200));
    w.bullets.push_back(w.bullets.front());
    w.enemies.push_back({{200, 200 - shooter::kEnemySpeed * shooter::kTickSeconds}, true});
    w.Tick({});
    Check(w.score == 100 && w.enemies.empty() && w.bullets.size() == 1,
          "two bullets cannot score twice against the same enemy");

    w.Reset();
    w.bullets.push_back(BulletAtAfterMove(320, 650));
    w.enemies.push_back(ContactEnemy(w));
    w.Tick({});
    Check(w.score == 100 && w.player.hp == 3 && w.enemies.empty(),
          "bullet collision is resolved before enemy/player contact");
}

void DamageFreezeAndRestart() {
    shooter::World w;
    w.enemies.push_back(ContactEnemy(w));
    w.enemies.push_back(w.enemies.front());
    w.Tick({});
    Check(w.player.hp == 2 && w.player.invulnerabilityTicks == 48 && w.enemies.empty(),
          "same-tick contacts cause one damage but remove all contacting enemies");
    for (int i = 0; i < 47; ++i) {
        w.enemies.push_back(ContactEnemy(w));
        w.Tick({});
        Check(w.player.hp == 2, "invulnerability prevents repeated damage");
    }
    w.enemies.push_back(ContactEnemy(w));
    w.Tick({});
    Check(w.player.hp == 1 && w.player.invulnerabilityTicks == 48, "damage allowed after 48 elapsed ticks");

    w.player.invulnerabilityTicks = 0; // Valid fixture immediately before lethal contact.
    w.bullets.push_back({{20, 300}, true});
    w.enemies.push_back(ContactEnemy(w));
    w.Tick({});
    Check(w.player.hp == 0 && w.gameOver, "HP zero enters GameOver");
    const float bulletY = w.bullets[0].position.y;
    const auto enemyCount = w.enemies.size();
    const auto bulletCount = w.bullets.size();
    const int spawnCountdown = w.spawnCountdownTicks;
    const int invulnerability = w.player.invulnerabilityTicks;
    const float playerX = w.player.position.x;
    Advance(w, 100, {1, 0, true, false});
    Check(w.bullets.size() == bulletCount && w.enemies.size() == enemyCount &&
          Near(w.bullets[0].position.y, bulletY) && Near(w.player.position.x, playerX) &&
          w.spawnCountdownTicks == spawnCountdown && w.player.invulnerabilityTicks == invulnerability,
          "GameOver freezes positions, spawn, shots and invulnerability");
    w.score = 900;
    w.Tick({1, 1, true, true});
    Check(w.player.hp == 3 && w.score == 0 && !w.gameOver && w.bullets.empty() && w.enemies.empty(),
          "restart resets gameplay state and returns without firing");
    Check(Near(w.player.position.x, 320) && Near(w.player.position.y, 650) &&
          w.shotCooldownTicks == 0 && w.spawnCountdownTicks == 39 && w.nextSpawnIndex == 0,
          "restart resets position and timers");
    w.Tick({0, 0, true, false});
    Check(w.bullets.size() == 1, "immediate first shot after restart");
}

void LimitsAndCleanup() {
    shooter::World w;
    for (std::size_t i = 0; i < shooter::kMaxBullets; ++i)
        w.bullets.push_back({{20, 400}, true});
    for (std::size_t i = 0; i < shooter::kMaxEnemies; ++i)
        w.enemies.push_back({{600, 100}, true});
    w.spawnCountdownTicks = 1;
    w.Tick({0, 0, true, false});
    Check(w.bullets.size() == 64 && w.enemies.size() == 32, "capacity limits skip additions");
    Check(w.shotCooldownTicks == 9 && w.spawnCountdownTicks == 39 && w.nextSpawnIndex == 1,
          "skipped additions still consume timers and spawn cycle");

    w.Reset();
    w.bullets.push_back({{10, -10}, true});
    w.bullets.push_back({{20, 100}, false});
    w.enemies.push_back({{100, 740}, true});
    w.enemies.push_back({{150, 100}, false});
    w.Tick({});
    Check(w.bullets.empty() && w.enemies.empty() && w.player.hp == 3 && w.score == 0,
          "offscreen/dead objects are removed without score or damage");
}

} // namespace

int main() {
    InitialMovementAndBounds();
    ShotAndSpawnTiming();
    CollisionOrderAndSingleHit();
    DamageFreezeAndRestart();
    LimitsAndCleanup();
    std::printf("PASS: %d shooter model checks\n", checks);
    return 0;
}
