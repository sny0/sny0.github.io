cmake_minimum_required(VERSION 3.20)
project(asset_lifecycle_model LANGUAGES CXX)
add_executable(asset_lifecycle_model main.cpp)
target_compile_features(asset_lifecycle_model PRIVATE cxx_std_20)
set_target_properties(asset_lifecycle_model PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(asset_lifecycle_model PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
else()
    target_compile_options(asset_lifecycle_model PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
