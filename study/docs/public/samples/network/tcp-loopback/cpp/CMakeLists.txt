cmake_minimum_required(VERSION 3.20)
project(tcp_loopback_observation LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This sample uses Windows Winsock 2.2.")
endif()
add_executable(tcp_loopback_observation main.cpp)
target_compile_features(tcp_loopback_observation PRIVATE cxx_std_20)
target_link_libraries(tcp_loopback_observation PRIVATE ws2_32)
set_target_properties(tcp_loopback_observation PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(tcp_loopback_observation PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
else()
    target_compile_options(tcp_loopback_observation PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
