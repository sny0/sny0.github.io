cmake_minimum_required(VERSION 3.20)
project(name_resolution_observation LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This sample uses Windows GetAddrInfoExW.")
endif()
add_executable(name_resolution_observation main.cpp)
target_compile_features(name_resolution_observation PRIVATE cxx_std_20)
target_link_libraries(name_resolution_observation PRIVATE ws2_32)
set_target_properties(name_resolution_observation PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(name_resolution_observation PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
else()
    target_compile_options(name_resolution_observation PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
