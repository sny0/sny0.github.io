cmake_minimum_required(VERSION 3.20)
project(integer_byte_observation LANGUAGES CXX)

add_executable(integer_byte_observation main.cpp)
target_compile_features(integer_byte_observation PRIVATE cxx_std_20)
set_target_properties(integer_byte_observation PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(integer_byte_observation PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
else()
    target_compile_options(integer_byte_observation PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
